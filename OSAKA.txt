    <script>
        let map;
        let markers = [];
        let currentType = 'normal';
        let editingId = null;
        let tempEditType = 'normal';
        const japanBounds = L.latLngBounds([20.0, 122.0], [46.0, 154.0]);

        window.onload = function() {
            map = L.map('map', { zoomControl: false, maxBounds: japanBounds, maxBoundsViscosity: 1.0, tap: true }).setView([34.6648, 135.4950], 13);
            L.tileLayer('https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png', { attribution: '' }).addTo(map);
            L.control.zoom({ position: 'bottomright' }).addTo(map);
        };

        function setType(type) {
            currentType = type;
            const btnNormal = document.getElementById('btnNormal');
            const btnStay = document.getElementById('btnStay');
            if (type === 'normal') {
                btnNormal.className = "type-toggle-btn px-4 py-1.5 rounded-lg text-xs bg-white text-slate-800 shadow";
                btnStay.className = "type-toggle-btn px-4 py-1.5 rounded-lg text-xs text-slate-500 hover:text-slate-700";
                btnStay.style.backgroundColor = ""; btnStay.style.color = "";
            } else {
                btnStay.className = "type-toggle-btn px-4 py-1.5 rounded-lg text-xs shadow-sm font-bold";
                btnStay.style.backgroundColor = "var(--color-yellow)";
                btnStay.style.color = "var(--color-coffee)";
                btnNormal.className = "type-toggle-btn px-4 py-1.5 rounded-lg text-xs text-slate-500 hover:text-slate-700";
                btnNormal.style.backgroundColor = ""; btnNormal.style.color = "";
            }
        }

        function toggleList() {
            const panel = document.getElementById('listPanel');
            const overlay = document.getElementById('overlay');
            const isActive = panel.classList.toggle('active');
            overlay.classList.toggle('hidden', !isActive);
        }

        function parseCoords(url) {
            const patterns = [/@(-?\d+\.\d+),(-?\d+\.\d+)/, /place\/.*?\/@(-?\d+\.\d+),(-?\d+\.\d+)/, /place\/(-?\d+\.\d+),(-?\d+\.\d+)/, /q=(-?\d+\.\d+),(-?\d+\.\d+)/, /^(-?\d+\.\d+)\s*,\s*(-?\d+\.\d+)$/];
            for (let p of patterns) {
                let match = url.match(p);
                if (match) return { lat: parseFloat(match[1]), lng: parseFloat(match[2]) };
            }
            return null;
        }

        function handleAdd() {
            const urlInput = document.getElementById('mapUrlInput');
            const nameInput = document.getElementById('locationNameInput');
            const coords = parseCoords(urlInput.value.trim());
            const name = nameInput.value.trim() || "未命名地點";

            if (!urlInput.value.trim()) return showStatus("請輸入網址");
            if (coords) {
                createMarker(coords.lat, coords.lng, name, urlInput.value.trim(), currentType);
                urlInput.value = ""; nameInput.value = "";
                showStatus(`已儲存：${name}`);
                if (currentType === 'stay') setType('normal');
            } else {
                showStatus("網址解析失敗");
            }
        }

        function createMarker(lat, lng, name, originalUrl, type) {
            const id = Date.now();
            const marker = L.marker([lat, lng], { 
                icon: L.divIcon({
                    className: 'custom-marker',
                    html: `<div class="${type === 'stay' ? 'stay-marker' : 'pulse-marker'}"></div>`,
                    iconSize: [20, 20],
                    iconAnchor: [10, 10]
                }) 
            }).addTo(map);

            updateMarkerPopup(marker, lat, lng, name, type);
            markers.push({ id, name, lat, lng, originalUrl, type, leafletMarker: marker });
            updateList();
            focusLocation(id);
        }

        function updateMarkerPopup(marker, lat, lng, name, type) {
            const navUrl = `https://www.google.com/maps/dir/?api=1&origin=My+Location&destination=${lat},${lng}`;
            marker.bindPopup(`
                <div class="p-4 flex flex-col items-center gap-4">
                    <span class="font-black text-black text-xl tracking-tight text-center">${name}</span>
                    <a href="${navUrl}" target="_blank" class="${type === 'stay' ? 'nav-link-stay' : 'nav-link-normal'}">
                       <svg xmlns="http://www.w3.org/2000/svg" class="h-5 w-5" fill="none" viewBox="0 0 24 24" stroke="currentColor" stroke-width="3"><path stroke-linecap="round" stroke-linejoin="round" d="M12 19l9 2-9-18-9 18 9-2zm0 0v-8" /></svg>
                       <span>開啟導航</span>
                    </a>
                </div>
            `, { closeButton: false, offset: [0, -10], minWidth: 200 });
        }

        function openEditModal(id, event) {
            if(event) event.stopPropagation();
            const m = markers.find(item => item.id === id);
            if (!m) return;
            editingId = id;
            document.getElementById('editName').value = m.name;
            document.getElementById('editUrl').value = m.originalUrl;
            setEditType(m.type);
            document.getElementById('editModal').classList.remove('hidden');
        }

        function closeEditModal() {
            document.getElementById('editModal').classList.add('hidden');
            editingId = null;
        }

        function setEditType(type) {
            tempEditType = type;
            const bNormal = document.getElementById('editTypeNormal');
            const bStay = document.getElementById('editTypeStay');
            if (type === 'normal') {
                bNormal.style.backgroundColor = "var(--color-deep-blue)";
                bNormal.style.color = "white";
                bNormal.style.borderColor = "var(--color-deep-blue)";
                bStay.style.backgroundColor = "transparent";
                bStay.style.color = "var(--color-coffee)";
                bStay.style.borderColor = "var(--color-coffee)";
            } else {
                bStay.style.backgroundColor = "var(--color-yellow)";
                bStay.style.color = "var(--color-coffee)";
                bStay.style.borderColor = "var(--color-coffee)";
                bNormal.style.backgroundColor = "transparent";
                bNormal.style.color = "var(--color-deep-blue)";
                bNormal.style.borderColor = "var(--color-deep-blue)";
            }
        }

        function saveEdit() {
            const m = markers.find(item => item.id === editingId);
            const newName = document.getElementById('editName').value.trim() || "未命名地點";
            const newUrl = document.getElementById('editUrl').value.trim();
            const coords = parseCoords(newUrl);

            if (!coords) return showStatus("無效的地圖網址");
            m.name = newName; m.originalUrl = newUrl; m.lat = coords.lat; m.lng = coords.lng; m.type = tempEditType;
            m.leafletMarker.setLatLng([m.lat, m.lng]);
            m.leafletMarker.setIcon(L.divIcon({
                className: 'custom-marker',
                html: `<div class="${m.type === 'stay' ? 'stay-marker' : 'pulse-marker'}"></div>`,
                iconSize: [20, 20],
                iconAnchor: [10, 10]
            }));
            updateMarkerPopup(m.leafletMarker, m.lat, m.lng, m.name, m.type);
            updateList(); closeEditModal(); showStatus("已更新地點資訊");
        }

        function updateList() {
            const listEl = document.getElementById('locationList');
            if (markers.length === 0) {
                listEl.innerHTML = `<div class="text-center py-20 text-slate-300 font-bold italic">目前沒有行程</div>`;
                return;
            }
            listEl.innerHTML = markers.map(m => `
                <li class="flex items-center justify-between p-4 bg-white border-2 border-black rounded-xl hover:shadow-xl active:scale-95 transition-all cursor-pointer group" onclick="focusLocation(${m.id})">
                    <div class="flex items-center gap-3 overflow-hidden">
                        <div class="w-3 h-3 rounded-full flex-shrink-0" style="background-color: ${m.type === 'stay' ? 'var(--color-yellow)' : 'var(--color-light-blue)'}; border: 2px solid black;"></div>
                        <span class="text-sm font-black truncate ${m.type === 'stay' ? 'text-[var(--color-coffee)]' : 'text-black'}">${m.name}</span>
                    </div>
                    <div class="flex items-center gap-2 flex-shrink-0">
                        <button onclick="openEditModal(${m.id}, event)" class="text-slate-400 hover:text-blue-600 transition-colors">
                            <svg xmlns="http://www.w3.org/2000/svg" class="h-5 w-5" fill="none" viewBox="0 0 24 24" stroke="currentColor" stroke-width="2.5"><path stroke-linecap="round" stroke-linejoin="round" d="M11 5H6a2 2 0 00-2 2v11a2 2 0 002 2h11a2 2 0 002-2v-5m-1.414-9.414a2 2 0 112.828 2.828L11.828 15H9v-2.828l8.586-8.586z" /></svg>
                        </button>
                        <button onclick="removeOne(${m.id}, event)" class="text-slate-400 hover:text-red-600 transition-colors">
                            <svg xmlns="http://www.w3.org/2000/svg" class="h-6 w-6" fill="none" viewBox="0 0 24 24" stroke="currentColor" stroke-width="2.5"><path stroke-linecap="round" stroke-linejoin="round" d="M6 18L18 6M6 6l12 12" /></svg>
                        </button>
                    </div>
                </li>
            `).join('');
        }

        function focusLocation(id) {
            const m = markers.find(item => item.id === id);
            if (!m) return;
            map.flyTo([m.lat, m.lng], 16, { animate: true, duration: 1.2 });
            setTimeout(() => { m.leafletMarker.openPopup(); }, 600);
            if (window.innerWidth < 768) toggleList();
        }

        function removeOne(id, event) {
            event.stopPropagation();
            const idx = markers.findIndex(m => m.id === id);
            if (idx > -1) {
                map.removeLayer(markers[idx].leafletMarker);
                markers.splice(idx, 1);
                updateList();
            }
        }

        function showStatus(text) {
            const msgBox = document.getElementById('statusMsg');
            document.getElementById('msgText').innerText = text;
            msgBox.classList.remove('hidden');
            setTimeout(() => msgBox.classList.add('hidden'), 3000);
        }

        document.getElementById('mapUrlInput').addEventListener('keypress', e => e.key === 'Enter' && handleAdd());
    </script>
